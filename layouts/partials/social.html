Missing 'status' path: %s" $url -}} {{- end -}} {{- if not $tweetid -}} {{- errorf "Invalid Twitter/X URL. Missing tweet id: %s" $url -}} {{- end -}} {{- $query := querify "url" (printf "https://twitter.com/%s/status/%s" $user $tweetid) -}} {{- $oembedUrl := printf "https://publish.twitter.com/oembed?%s" $query -}} {{- partial "social-fetch.html" $oembedUrl -}} {{- else if in $url "bsky.app" -}} {{- $params := dict "url" $url "format" "json" -}} {{- $queryString := "" -}} {{- range $k, $v := $params -}} {{- if eq $queryString "" -}} {{- $queryString = printf "%s=%s" $k (urlquery $v) -}} {{- else -}} {{- $queryString = printf "%s&%s=%s" $queryString $k (urlquery $v) -}} {{- end -}} {{- end -}} {{- $bskyReq := printf "https://embed.bsky.app/oembed?%s" $queryString -}} {{- partial "social-fetch.html" $bskyReq -}} {{- else if in $url "linkedin.com" -}}
